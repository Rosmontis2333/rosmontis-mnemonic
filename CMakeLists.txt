cmake_minimum_required(VERSION 3.20)
project(rosmontis_mnemonic C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only 64-bit builds are supported (8 GiB working memory).")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ARGON argon2/src)
add_executable(rosmontis-mnemonic main.c sha2.c
  ${ARGON}/argon2.c ${ARGON}/core.c ${ARGON}/encoding.c
  ${ARGON}/ref.c ${ARGON}/thread.c ${ARGON}/blake2/blake2b.c)
target_include_directories(rosmontis-mnemonic PRIVATE . argon2/include ${ARGON})
target_compile_definitions(rosmontis-mnemonic PRIVATE ROSMONTIS_PROGRESS)
if(WIN32)
  target_sources(rosmontis-mnemonic PRIVATE platform_windows.c)
  target_compile_definitions(rosmontis-mnemonic PRIVATE _WIN32_WINNT=0x0A00)
else()
  find_package(Threads REQUIRED)
  target_sources(rosmontis-mnemonic PRIVATE platform_posix.c)
  target_link_libraries(rosmontis-mnemonic PRIVATE Threads::Threads)
endif()
if(MSVC)
  target_compile_options(rosmontis-mnemonic PRIVATE /O2 /utf-8 /W4)
  target_compile_definitions(rosmontis-mnemonic PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(rosmontis-mnemonic PRIVATE -O2 -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(rosmontis-selftest tests/selftest.c sha2.c
    ${ARGON}/argon2.c ${ARGON}/core.c ${ARGON}/encoding.c
    ${ARGON}/ref.c ${ARGON}/thread.c ${ARGON}/blake2/blake2b.c)
  target_include_directories(rosmontis-selftest PRIVATE . tests argon2/include ${ARGON})
  target_compile_definitions(rosmontis-selftest PRIVATE ROSMONTIS_PROGRESS)
  if(WIN32)
    target_sources(rosmontis-selftest PRIVATE platform_windows.c)
    target_compile_definitions(rosmontis-selftest PRIVATE _WIN32_WINNT=0x0A00)
  else()
    target_sources(rosmontis-selftest PRIVATE platform_posix.c)
    target_link_libraries(rosmontis-selftest PRIVATE Threads::Threads)
  endif()
  if(MSVC)
    target_compile_options(rosmontis-selftest PRIVATE /O2 /utf-8)
  endif()
  add_test(NAME crypto_and_secure_memory COMMAND rosmontis-selftest)
endif()

if(APPLE)
  find_program(CODESIGN codesign REQUIRED)
  add_custom_command(TARGET rosmontis-mnemonic POST_BUILD
    COMMAND "${CODESIGN}" --force --sign - "$<TARGET_FILE:rosmontis-mnemonic>")
  if(BUILD_TESTING)
    add_custom_command(TARGET rosmontis-selftest POST_BUILD
      COMMAND "${CODESIGN}" --force --sign - "$<TARGET_FILE:rosmontis-selftest>")
  endif()
endif()

set(WALLET_SOURCES wallet.c keccak.c vendor/ripemd160.c
  vendor/secp256k1-0.8.0/src/secp256k1.c
  vendor/secp256k1-0.8.0/src/precomputed_ecmult.c
  vendor/secp256k1-0.8.0/src/precomputed_ecmult_gen.c)
foreach(target rosmontis-mnemonic rosmontis-selftest)
  if(TARGET ${target})
    target_sources(${target} PRIVATE ${WALLET_SOURCES})
    target_compile_definitions(${target} PRIVATE SECP256K1_STATIC)
    target_include_directories(${target} PRIVATE vendor vendor/secp256k1-0.8.0/include)
    if(WIN32)
      target_link_libraries(${target} PRIVATE bcrypt)
    endif()
  endif()
endforeach()

if(NOT MSVC)
  set_source_files_properties(vendor/secp256k1-0.8.0/src/secp256k1.c PROPERTIES COMPILE_OPTIONS "-Wno-unused-function")
endif()
